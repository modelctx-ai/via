import type { NextFunction, Request, Response } from 'express';
import { redis } from '../lib/redis';
import { limits } from '../config/limits';
import { logger } from '../lib/logger';

export interface RateLimiterOptions {
  /** Requests allowed per window. */
  max: number;
  /** Window length in seconds. */
  windowSeconds: number;
  /** Derives the bucket key; defaults to the client IP. */
  key?: (req: Request) => string;
}

/**
 * Fixed-window rate limiter backed by Redis. When Redis is unreachable the limiter fails open
 * and logs a warning, so an outage never takes the API down with it.
 */
export class RateLimiter {
  constructor(private readonly options: RateLimiterOptions = limits.default) {}

  middleware = async (req: Request, res: Response, next: NextFunction): Promise<void> => {
    const key = `rl:${(this.options.key ?? defaultKey)(req)}`;
    try {
      const count = await redis.incr(key);
      if (count === 1) await redis.expire(key, this.options.windowSeconds);
      res.setHeader('X-RateLimit-Limit', String(this.options.max));
      res.setHeader('X-RateLimit-Remaining', String(Math.max(0, this.options.max - count)));
      if (count > this.options.max) {
        const retryAfter = await redis.ttl(key);
        res.setHeader('Retry-After', String(Math.max(retryAfter, 1)));
        res.status(429).json({ error: 'rate_limited', retryAfter });
        return;
      }
    } catch (error) {
      logger.warn({ error }, 'rate limiter unavailable, failing open');
    }
    next();
  };
}

function defaultKey(req: Request): string {
  return req.ip ?? 'unknown';
}

export const apiLimiter = new RateLimiter();
